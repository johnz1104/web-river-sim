# WebAssembly build of the river for the website. A separate project from
# fluids/CMakeLists.txt; configure it with Emscripten (`npm run build:wasm`):
#   emcmake cmake -S fluids/lbm/wasm -B fluids/build/wasm -DCMAKE_BUILD_TYPE=Release
#   cmake --build fluids/build/wasm
# The output, lbm.js and lbm.wasm, is copied to src/river/sim/generated/ and
# committed, so the site builds without Emscripten.
cmake_minimum_required(VERSION 3.14)
project(lbm_wasm LANGUAGES CXX)

if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "Configure this project with emcmake (see the header).")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(${CMAKE_CURRENT_LIST_DIR}/../sources.cmake)

option(LBM_WASM_SIMD "Compile with -msimd128" OFF)
# Memory the module starts with, in bytes (a multiple of 64 KiB). It grows on demand
# (ALLOW_MEMORY_GROWTH), so this only sets what is reserved up front: Emscripten's
# default is 16 MiB; the page river peaks at 7.9 MB (DECISIONS.md, "memory").
set(LBM_WASM_INITIAL_MEMORY 8388608 CACHE STRING "Initial WebAssembly memory in bytes")
set(LBM_WASM_OUTPUT_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../src/river/sim/generated"
    CACHE PATH "Where lbm.js and lbm.wasm are copied")

add_executable(lbm bindings.cpp ${LBM_CORE_SOURCES})
target_include_directories(lbm PRIVATE ${LBM_INCLUDE_DIR})
# Single precision, as validated natively by lbm_river_f32. Never -ffast-math: it lets
# the compiler reorder float sums and assume no NaN, which would fold away the
# diagnostics' finite-state check.
target_compile_definitions(lbm PRIVATE LBM_SINGLE_PRECISION)
# Wasm exceptions, so a bad option or bank reports an error instead of aborting.
set(LBM_WASM_FLAGS -O3 -fwasm-exceptions)
if(LBM_WASM_SIMD)
    list(APPEND LBM_WASM_FLAGS -msimd128)
endif()
target_compile_options(lbm PRIVATE ${LBM_WASM_FLAGS})

set(LBM_WASM_EXPORTS
    _river_create _river_last_error _river_set_view _river_warm_up _river_export_flow
    _river_start_from_flow _river_advance _river_dot_count
    _river_dots _river_dot_stride _river_tail_ages _river_set_dot_jitter
    _river_diagnostic_count _river_diagnostics _river_destroy _malloc _free)
list(JOIN LBM_WASM_EXPORTS "," LBM_WASM_EXPORTS)
target_link_options(lbm PRIVATE
    ${LBM_WASM_FLAGS}
    -sMODULARIZE
    -sEXPORT_ES6
    -sEXPORT_NAME=createLbmModule
    -sENVIRONMENT=worker,node
    -sALLOW_MEMORY_GROWTH
    -sINITIAL_MEMORY=${LBM_WASM_INITIAL_MEMORY}
    -sFILESYSTEM=0
    -sEXPORTED_FUNCTIONS=${LBM_WASM_EXPORTS}
    -sEXPORTED_RUNTIME_METHODS=HEAPU8,HEAPF32,HEAPF64,stringToUTF8,lengthBytesUTF8,UTF8ToString
)
set_target_properties(lbm PROPERTIES SUFFIX ".js")

# A banner on lbm.js; it also keeps the linter off Emscripten's generated glue.
set(LBM_WASM_BANNER ${CMAKE_CURRENT_BINARY_DIR}/banner.js)
file(WRITE ${LBM_WASM_BANNER}
    "// Generated by `npm run build:wasm` (fluids/lbm/wasm/CMakeLists.txt). Do not edit.\n"
    "/* eslint-disable */\n")
target_link_options(lbm PRIVATE "SHELL:--extern-pre-js ${LBM_WASM_BANNER}")
set_target_properties(lbm PROPERTIES LINK_DEPENDS ${LBM_WASM_BANNER})

# Copies on every build, not only after a relink, so the committed files always
# match this build tree.
add_custom_target(lbm_output ALL DEPENDS lbm
    COMMAND ${CMAKE_COMMAND} -E make_directory ${LBM_WASM_OUTPUT_DIR}
    COMMAND ${CMAKE_COMMAND} -E copy
            ${CMAKE_CURRENT_BINARY_DIR}/lbm.js ${CMAKE_CURRENT_BINARY_DIR}/lbm.wasm
            ${LBM_WASM_OUTPUT_DIR}
    COMMENT "Copying lbm.js and lbm.wasm to ${LBM_WASM_OUTPUT_DIR}")
